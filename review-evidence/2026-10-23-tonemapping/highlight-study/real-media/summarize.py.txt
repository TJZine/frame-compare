"""Build auditable summaries and a locally viewable comparison; no native imports."""
import hashlib,html,json,re
from pathlib import Path
import numpy as np
from PIL import Image, ImageDraw
from probe import ROOT,source_identity,dump,sha

rows=[]
for p in sorted(ROOT.glob('stage*/**/result.json')):
    d=json.loads(p.read_text()); d['result_path']=str(p.relative_to(ROOT))
    d['image_path']=str((p.parent/'final.png').relative_to(ROOT)).replace('\\','/')
    text=(p.parent/'stdout.txt').read_text()+'\n'+(p.parent/'stderr.txt').read_text()
    traces=[line for line in text.splitlines() if re.search(r'tone map \(|[Pp]eak|[Ll]uminance|Device Name:|Driver info:',line)]
    (p.parent/'peak-trace.txt').write_text('\n'.join(traces)+'\n')
    d['effective_peak_trace']=[x for x in traces if 'tone map (' in x]
    d['effective_peak_status']='observed trace' if d['effective_peak_trace'] else 'unavailable: verbose native logs contain shader source but no per-frame effective-peak receipt; supplied src_max is not substituted'
    for plane in ['rgb8','native16']:
        if plane+'_metrics' in d:
            d[plane+'_picture']=d[plane+'_metrics']['picture']
    rows.append(d)
by={(r['scene'],r['variant']):r for r in rows}
baseline=by['porch','baseline'];plain=by['porch','plain']
assert baseline['rgb8_sha256']==plain['rgb8_sha256']
checks={'tap_equals_uninstrumented':True,'source_stable':all(r['source_identity']==rows[0]['source_identity'] for r in rows),
        'all_original_indexes_unchanged':all(r['index']['before']==r['index']['after'] for r in rows),
        'all_scratch_removed':all(json.loads((ROOT/r['result_path']).with_name('cleanup.json').read_text())['removed'] for r in rows)}
diffs=[]
for scene in sorted({r['scene'] for r in rows}):
    base=by.get((scene,'baseline'))
    if base is None:continue
    b=np.asarray(Image.open(ROOT/base['image_path'])).astype(np.int16)
    for r in rows:
        if r['scene']!=scene or r['variant']=='baseline':continue
        a=np.asarray(Image.open(ROOT/r['image_path'])).astype(np.int16); delta=a-b
        diffs.append({'scene':scene,'variant':r['variant'],'baseline':'baseline','mean_abs_code8':float(np.abs(delta).mean()),
          'max_abs_code8':int(np.abs(delta).max()),'changed_components_pct':float((delta!=0).mean()*100),
          'mean_channel_delta':delta.mean(axis=(0,1)).tolist()})
compact=[{k:r[k] for k in ['scene','variant','source_frame','source_frame_kind','source_time','settings','rgb8_sha256','result_path','image_path','rgb8_picture','effective_peak_status']}|
         {k:r[k] for k in ['native16_sha256','native16_picture'] if k in r} for r in rows]
dump(ROOT/'summary.json',{'source':source_identity(),'checks':checks,'case_count':len(rows),'cases':compact,'differentials':diffs,
   'limits':['Windows unmanaged native profile only; no calibrated display or SDR grade golden',
             'Metrics use fixed stride4 coordinates below y250; exact bounds retained per result',
             'Source identity and source packet/profile provenance retained; historical packet-copy equality not freshly rerun',
             'Effective peak numeric trace unavailable in fresh still graphs despite log_level6; metadata4 branch backed by pinned source and stripped input props',
             'No product changes or default recommendation accepted by this executor']})
parts=['<!doctype html><meta charset="utf-8"><title>Current native highlight study — Unit A</title><style>body{background:#161616;color:#eee;font:16px system-ui;margin:24px}h1{font-size:26px}section{display:grid;grid-template-columns:repeat(2,minmax(350px,1fr));gap:16px}figure{margin:0;background:#222;padding:12px}img{width:100%}pre{white-space:pre-wrap;font-size:12px}a{color:#aaf}</style><h1>Current native highlight study — Unit A</h1><p>100-nit target, corrected sRGB, independent quality stills. Labels identify explicit diagnostic changes. Metrics are sampled code values, not fidelity scores. Click each image for full size. <a href="summary.json">JSON summary</a></p>']
for scene in ['porch','closeup','dark','spiderverse','substance','immaculate']:
    scene_rows=[r for r in rows if r['scene']==scene and r['variant']!='plain'];parts.append('<h2>'+scene+'</h2><section>')
    thumbs=[]
    for r in scene_rows:
        m=r['rgb8_picture'];label=f"{scene} frame {r['source_frame']} | {r['variant']} | all255 {m['all_endpoint_pct']:.4f}% | any255 {m['any_endpoint_pct']:.4f}% | mean {m['mean_code8']:.2f}"
        href=html.escape(r['image_path']);parts.append(f'<figure><figcaption>{html.escape(label)}</figcaption><a href="{href}"><img src="{href}"></a><a href="{html.escape(r["result_path"].replace(chr(92),"/"))}">Full receipt / settings / regions</a></figure>')
        im=Image.open(ROOT/r['image_path']).convert('RGB');im.thumbnail((760,420));thumb=Image.new('RGB',(780,460),'#222222');thumb.paste(im,(10,35));ImageDraw.Draw(thumb).text((10,10),label,fill='white');thumbs.append(thumb)
    parts.append('</section>')
    sheet=Image.new('RGB',(780*2,460*((len(thumbs)+1)//2)), '#111111')
    for i,t in enumerate(thumbs):sheet.paste(t,((i%2)*780,(i//2)*460))
    sheet.save(ROOT/(scene+'-labeled.png'))
(ROOT/'comparison.html').write_text(''.join(parts),encoding='utf-8')
dump(ROOT/'artifact-manifest.json',{str(p.relative_to(ROOT)):sha(p) for p in ROOT.rglob('*') if p.is_file() and p.name!='artifact-manifest.json' and '__pycache__' not in p.parts})
print(json.dumps({'cases':len(rows),'checks':checks}))
