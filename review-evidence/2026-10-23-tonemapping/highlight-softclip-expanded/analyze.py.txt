def calc(b,k):
    delta=k.astype(float)-b.astype(float); yb=b@np.array([.2126,.7152,.0722]);yk=k@np.array([.2126,.7152,.0722])

      'max_abs_delta_code8':float(np.abs(delta).max()),'p99_abs_delta_code8':float(np.percentile(np.abs(delta),99)),
      'components_delta_at_least2_pct':float((np.abs(delta)>=2).mean()*100),
      'mean_luma_delta_code8':float((yk-yb).mean()),
      'mean_chroma_delta_code8':float((np.ptp(k.astype(float),axis=2)-np.ptp(b.astype(float),axis=2)).mean()),
      'perceptual':{'any255_pct':float(np.any(b==255,axis=2).mean()*100),'all255_pct':float(np.all(b==255,axis=2).mean()*100),
        'luma_gradient':float(np.abs(np.diff(yb,axis=1)).mean()),'mean_code8':float(b.mean())},
      'softclip':{'any255_pct':float(np.any(k==255,axis=2).mean()*100),'all255_pct':float(np.all(k==255,axis=2).mean()*100),
