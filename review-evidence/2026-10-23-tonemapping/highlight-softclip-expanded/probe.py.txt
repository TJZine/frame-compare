"""Owned diagnostic only. Fresh child per case, same-evaluation native/output taps."""
from dataclasses import asdict, replace
import argparse, datetime, hashlib, json, os, platform, shutil, subprocess, sys, tempfile
from pathlib import Path

ROOT=Path(__file__).resolve().parent
REPO=next(p for p in ROOT.parents if (p/'.git').exists())
HIST=REPO/'generated/verification/2026-10-20/hdr-color-study/cross-title'

def sha(path):
    with Path(path).open('rb') as f: return hashlib.file_digest(f,'sha256').hexdigest()
def dump(path, obj): Path(path).write_text(json.dumps(obj,indent=2,default=str)+'\n',encoding='utf-8')
def source_identity():
    return {'sha':subprocess.check_output(['git','rev-parse','HEAD'],cwd=REPO,text=True).strip(),
      'status':subprocess.check_output(['git','status','--short'],cwd=REPO,text=True),
      'files':{str(p.relative_to(REPO)):sha(p) for p in [*sorted((REPO/'src/frame_compare/vs').glob('tonemap*.py')),REPO/'src/frame_compare/vs/types.py',REPO/'src/frame_compare/render/encoders.py',REPO/'src/frame_compare/render/prepare.py',REPO/'src/frame_compare/config/schema_enums.py',REPO/'config/config.toml']}}

def worker(scene,variant,out):
    source_start=source_identity()
    import numpy as np
    from PIL import Image, PngImagePlugin
    from frame_compare.vs.env import ensure_vs_environment
    from frame_compare.vs.source import source_index_path
    from frame_compare.vs import tonemap_libplacebo as tm
    from frame_compare.vs.types import TonemapSettings, TonemapResult
    from frame_compare.config.schema import ToneCurve
    from frame_compare.render.encoders import _clip_to_rgb24_for_pillow
    core=ensure_vs_environment(); core.num_threads=4; core.max_cache_size=128
    logs=[]; handler=core.add_log_handler(lambda level,message:logs.append({'level':int(level),'message':message}))
    def props(p):
        return {k:({'bytes':len(v),'sha256':hashlib.sha256(v).hexdigest()} if isinstance(v,bytes) else v) for k,v in dict(p).items()}
    def arr(f): return np.stack([np.asarray(f[i]) for i in range(3)],axis=2).copy()
    def ah(a): return hashlib.sha256(a.tobytes()).hexdigest()
    def metrics(a,maximum):
        # Full picture samples below the historical top-overlay boundary; fixed coordinate patches.
        h,w=a.shape[:2]
        regions={'picture':[0,250,w,h], 'upper_center':[int(w*.35),int(h*.20),int(w*.65),int(h*.45)],
                 'left_middle':[int(w*.05),int(h*.35),int(w*.30),int(h*.65)],
                 'center':[int(w*.35),int(h*.40),int(w*.65),int(h*.70)]}
        result={}
        for name,(x0,y0,x1,y1) in regions.items():
            r=a[y0:y1:4,x0:x1:4]; f=r.astype(float)/maximum; y=f@np.array([.2126,.7152,.0722])
            result[name]={'bounds_xyxy':[x0,y0,x1,y1],'stride':4,'pixels':int(y.size),
                'mean_code8':float(f.mean()*255),'all_endpoint_pct':float(np.all(r>=maximum,axis=2).mean()*100),
                'any_endpoint_pct':float(np.any(r>=maximum,axis=2).mean()*100),
                'all_near_white_pct':float(np.all(f>=250/255,axis=2).mean()*100),
                'luma_code8_std':float(y.std()*255),'horizontal_gradient_code8':float(np.abs(np.diff(y,axis=1)).mean()*255),
                'unique_rgb_triplets':int(len(np.unique(r.reshape(-1,3),axis=0)))}
        return result
    settings=TonemapSettings()
    if variant=='configured':
        from frame_compare.render.prepare import resolve_tonemap_settings
        from frame_compare.config.schema import ConfigSchema,ColorConfig,TonemapPreset
        settings=resolve_tonemap_settings(ConfigSchema(color=ColorConfig(preset=TonemapPreset.HIGHLIGHT_PRESERVE)))
    if variant.startswith('knee1'): settings=replace(settings,knee_offset=1.)
    if variant=='spline': settings=replace(settings,tone_curve=ToneCurve.SPLINE,knee_offset=.5)
    if variant=='metadata2': settings=replace(settings,metadata=2)
    if variant.startswith('measured'): settings=replace(settings,use_dovi=False,metadata=4,percentile=100. if variant=='measured100' else 99.995)
    if variant=='static448': settings=replace(settings,use_dovi=False,metadata=2,source_peak=448,dynamic_peak_detection=False)
    if variant.endswith('gamut2'): settings=replace(settings,gamut_mapping=2)
    if variant.endswith('gamut7'): settings=replace(settings,gamut_mapping=7)
    if variant.endswith('contrast0'): settings=replace(settings,contrast_recovery=0.)
    provenance=None
    if scene in ['porch','closeup','dark']:
        media=next((REPO/'comparison_videos').glob('Kill Bill*.mkv')); n={'porch':12382,'closeup':14943,'dark':110257}[scene]
    else:
        title,scene_id,frame_id=scene.split('-');scene_id=int(scene_id);n=int(frame_id)
        key=title+'-hdr-'+str(scene_id)
        samples=json.loads((HIST/'samples.json').read_text())['samples']
        provenance=next(x for x in samples if x['role']==title+'-hdr' and x['scene']==scene_id)
        media=Path(provenance['path']); assert sha(media)==provenance['sha256']
    original_index=source_index_path(media)
    if not original_index.exists():
        candidates=list(media.parent.glob(media.name+'*.lwi'))
        original_index=candidates[0] if candidates else None
    index_before=sha(original_index) if original_index else None
    scratch=Path(tempfile.mkdtemp(prefix='frame-compare-highlight-A-'))
    marker={'pid':os.getpid(),'owner':str(ROOT),'scratch':str(scratch)}
    dump(scratch/'owned.json',marker);dump(out/'scratch.json',marker)
    try:
        copied=scratch/'source.lwi'
        if original_index:shutil.copy2(original_index,copied)
        source=core.lsmas.LWLibavSource(str(media),cachefile=str(copied),threads=4)
        source_frame=source.get_frame(n); source_props=props(source_frame.props); target_time=source_frame.props['_AbsoluteTime']; del source_frame
        removed=[]
        if variant.startswith('measured') or variant=='static448':
            removed=['DolbyVisionRPU','PLSceneMax','PLSceneAvg']
            source=source.std.RemoveFrameProps(props=removed)
        if variant=='static448':
            source=source.std.SetFrameProps(ContentLightLevelMax=448.,ContentLightLevelAverage=92.,MasteringDisplayMaxLuminance=448.)
        captures={}; calls=[]
        class Placebo:
            def Tonemap(self,clip,**kwargs):
                kwargs=dict(kwargs); kwargs['log_level']=6; calls.append(kwargs)
                native=core.placebo.Tonemap(clip,**kwargs)
                def tap(n,f):
                    if abs(f.props['_AbsoluteTime']-target_time)<1e-6:
                        captures['native16']=arr(f); captures['native_props']=props(f.props)
                        inp=clip.get_frame(n); captures['input_props']=props(inp.props); captures['input_rgb16_sha256']=ah(arr(inp));del inp
                    return f
                return native.std.ModifyFrame(clips=native,selector=tap)
        class Proxy:
            placebo=Placebo()
            def __getattr__(self,name):return getattr(core,name)
        plain=variant.endswith('plain')
        result=tm.apply_libplacebo(source,settings,core if plain else Proxy(),still_image=True)
        assert isinstance(result,TonemapResult)
        rendered=result.clip
        if not plain:
            def tap_srgb(n,f):
                if n==target_n: captures['srgb_float']=arr(f)
                return f
            target_n=n
            rendered=rendered.std.ModifyFrame(clips=rendered,selector=tap_srgb)
        rgb8=arr(_clip_to_rgb24_for_pillow(rendered).get_frame(n))
        pnginfo=PngImagePlugin.PngInfo();pnginfo.add(b'sRGB',b'\x00')
        Image.fromarray(rgb8).save(out/'final.png',pnginfo=pnginfo)
        row={'scene':scene,'variant':variant,'source':str(media),'source_frame':n,'source_frame_kind':'original' if provenance is None else 'retained_excerpt',
          'provenance':provenance,'source_props':source_props,'source_time':target_time,'removed_props':removed,'settings':asdict(settings),
          'applied':asdict(result.applied),'calls':calls,'rgb8_sha256':ah(rgb8),'png_sha256':sha(out/'final.png'),
          'rgb8_metrics':metrics(rgb8,255),'source_identity':source_identity(),'source_identity_start':source_start,
          'runtime':{'python':sys.version,'platform':platform.platform(),'core':str(core.version()),'tonemap_signature':str(core.placebo.Tonemap.signature),
          'plugins':[{'namespace':p.namespace,'name':p.name,'identifier':p.identifier,'path':p.plugin_path,'sha256':sha(p.plugin_path) if p.plugin_path and Path(p.plugin_path).is_file() else None} for p in core.plugins() if p.namespace in ['placebo','lsmas']]},
          'index':{'original':str(original_index),'before':index_before,'after':sha(original_index) if original_index else None,'scratch':str(copied)},
          'capture_boundary':'native ModifyFrame and sRGB float taps observed upstream of the SAME final RGB8 get_frame; no second native get_frame' if not plain else 'uninstrumented product apply_libplacebo + Pillow RGB24 helper'}
        if not plain:
            assert 'native16' in captures and 'srgb_float' in captures
            raw=captures.pop('native16');srgb=captures.pop('srgb_float')
            np.savez_compressed(out/'native16.npz',rgb16=raw)
            row.update(captures);row.update(native16_sha256=ah(raw),srgb_float_sha256=ah(srgb),native16_metrics=metrics(raw,65535),srgb_float_metrics=metrics(srgb,1))
        assert row['index']['before']==row['index']['after']
        dump(out/'result.json',row)
        dump(out/'native-logs.json',logs)
        (out/'peak-trace.txt').write_text('\n'.join(x['message'] for x in logs if any(s in x['message'].lower() for s in ['tone map','tone-map','peak','luminance','vulkan device']))+'\n',encoding='utf-8')
        core.remove_log_handler(handler)
        print(json.dumps({'case':out.name,'rgb8':row['rgb8_sha256'],'metrics':row['rgb8_metrics']['picture']}),flush=True)
    finally:
        dump(out/'native-logs.json',logs)
        assert scratch.resolve().parent==Path(tempfile.gettempdir()).resolve() and json.loads((scratch/'owned.json').read_text())==marker
        assert not any(p.is_symlink() or p.is_junction() for p in scratch.rglob('*'))
        shutil.rmtree(scratch);dump(out/'cleanup.json',{'scratch':str(scratch),'removed':not scratch.exists()})

def run(cases,out):
    out.mkdir(parents=True,exist_ok=True); receipts=[]
    for case in cases:
        scene,variant=case.split(':');caseout=out/(scene+'-'+variant);caseout.mkdir(exist_ok=False)
        args=[sys.executable,str(Path(__file__).resolve()),'worker',case,'--out',str(caseout)]
        started=datetime.datetime.now(datetime.timezone.utc).isoformat()
        try:
            p=subprocess.run(args,capture_output=True,text=True,timeout=150,cwd=REPO)
            code=p.returncode;stdout=p.stdout;stderr=p.stderr
        except subprocess.TimeoutExpired as e:
            code='timeout';stdout=str(e.stdout);stderr=str(e.stderr)
            # subprocess.run has killed and reaped its owned child before raising.
            marker_path=caseout/'scratch.json'
            if marker_path.exists():
                marker=json.loads(marker_path.read_text());scratch=Path(marker['scratch'])
                assert marker['owner']==str(ROOT) and scratch.resolve().parent==Path(tempfile.gettempdir()).resolve()
                assert scratch.name.startswith('frame-compare-highlight-A-')
                assert json.loads((scratch/'owned.json').read_text())==marker
                assert not any(p.is_symlink() or p.is_junction() for p in scratch.rglob('*'))
                shutil.rmtree(scratch);dump(caseout/'cleanup.json',{'scratch':str(scratch),'removed':not scratch.exists(),'after_timeout':True})
        (caseout/'stdout.txt').write_text(stdout,encoding='utf-8');(caseout/'stderr.txt').write_text(stderr,encoding='utf-8')
        (caseout/'peak-trace.txt').write_text('\n'.join(line for line in (stdout+'\n'+stderr).splitlines() if any(s in line.lower() for s in ['tone mapping','tone-mapping','peak','luminance','device name:','driver info:']))+'\n',encoding='utf-8')
        receipts.append({'command':args,'started':started,'exit':code,'reaped':True});dump(out/'invocations.json',receipts)
        print(case,code,flush=True)
        if code!=0:raise SystemExit(1)

if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('mode',choices=['worker','run']);p.add_argument('cases',nargs='+');p.add_argument('--out',type=Path,required=True);a=p.parse_args()
    if a.mode=='worker':worker(*a.cases[0].split(':'),a.out)
    else:run(a.cases,a.out)
